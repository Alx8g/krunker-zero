cmake_minimum_required(VERSION 3.20)
# Set the CRT before targets are created. The supplied Windows source-build recipe
# produces release /MT libraries. Overrides must match the chosen V8 SDK exactly.
if(NOT DEFINED CMAKE_MSVC_RUNTIME_LIBRARY)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>" CACHE STRING "MSVC runtime matching the V8 SDK")
endif()
project(krunker_zero VERSION 0.4.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
find_package(Threads REQUIRED)

function(zero_compiler_options target)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /utf-8 /permissive- /Zc:__cplusplus /EHsc)
    target_compile_definitions(${target} PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra)
  endif()
endfunction()
add_library(zero_scheduler native/task_queue.cc)
target_include_directories(zero_scheduler PUBLIC native)
target_link_libraries(zero_scheduler PUBLIC Threads::Threads)
zero_compiler_options(zero_scheduler)
add_library(zero_platform native/platform.cc)
target_include_directories(zero_platform PUBLIC native)
if(CMAKE_DL_LIBS)
  target_link_libraries(zero_platform PRIVATE ${CMAKE_DL_LIBS})
endif()
zero_compiler_options(zero_platform)
if(BUILD_TESTING)
  add_executable(zero_scheduler_tests tests/task_queue_test.cc)
  target_link_libraries(zero_scheduler_tests PRIVATE zero_scheduler)
  zero_compiler_options(zero_scheduler_tests)
  add_test(NAME native_scheduler COMMAND zero_scheduler_tests)
  add_library(zero_platform_fixture SHARED tests/platform_fixture.cc)
  zero_compiler_options(zero_platform_fixture)
  add_executable(zero_platform_tests tests/platform_test.cc)
  target_link_libraries(zero_platform_tests PRIVATE zero_platform)
  add_dependencies(zero_platform_tests zero_platform_fixture)
  zero_compiler_options(zero_platform_tests)
  add_test(NAME native_platform COMMAND zero_platform_tests)
endif()

# Shipping target: standalone V8 only. No libnode, browser, or WebView fallback.
option(ZERO_BUILD_V8_HOST "Build standalone browserless V8 executable" OFF)
option(ZERO_BUILD_GRAPHICS "Build experimental EGL graphics (Linux or Windows ANGLE)" OFF)
set(ZERO_V8_ROOT "" CACHE PATH "SDK containing include/ and lib/")
set(ZERO_V8_INCLUDE_DIR "" CACHE PATH "V8 public include directory")
set(ZERO_V8_LIBRARY "" CACHE FILEPATH "Matching v8_monolith.lib or libv8_monolith.a")
set(ZERO_V8_EXTRA_LIBRARIES "" CACHE STRING "Additional matching V8 archives if required")
set(ZERO_V8_ICU_DATA "" CACHE FILEPATH "Matching icudtl.dat when ICU uses external data")
set(ZERO_V8_DEFINITIONS "" CACHE STRING "ABI defines matching V8; prefer its generated v8-gn.h")
set(ZERO_ANGLE_DIR "" CACHE PATH "Windows directory containing matching standalone ANGLE DLLs")
if(ZERO_V8_ROOT)
  set(ZERO_V8_INCLUDE_DIR "${ZERO_V8_ROOT}/include")
  if(WIN32)
    set(ZERO_V8_LIBRARY "${ZERO_V8_ROOT}/lib/v8_monolith.lib")
  else()
    set(ZERO_V8_LIBRARY "${ZERO_V8_ROOT}/lib/libv8_monolith.a")
  endif()
endif()
if(ZERO_BUILD_V8_HOST)
  if(WIN32 AND (NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8))
    message(FATAL_ERROR "Windows host target is MSVC-compatible x64 only. Use the x64 Native Tools prompt and the supplied Windows SDK recipe.")
  endif()
  if(WIN32 AND (CMAKE_GENERATOR_PLATFORM MATCHES "[Aa][Rr][Mm]" OR CMAKE_SYSTEM_PROCESSOR MATCHES "[Aa][Rr][Mm]"))
    message(FATAL_ERROR "Windows ARM64 is not a supported target for this SDK recipe; use x64.")
  endif()
  if(NOT EXISTS "${ZERO_V8_INCLUDE_DIR}/v8.h" OR NOT EXISTS "${ZERO_V8_LIBRARY}")
    message(FATAL_ERROR "Supply ZERO_V8_ROOT or ZERO_V8_INCLUDE_DIR + ZERO_V8_LIBRARY from the SAME standalone SDK. See docs/WINDOWS.md or docs/BUILD.md.")
  endif()
  if(WIN32 AND NOT EXISTS "${ZERO_V8_INCLUDE_DIR}/v8-gn.h" AND NOT ZERO_V8_DEFINITIONS)
    message(FATAL_ERROR "Windows V8 needs matching v8-gn.h or explicit ZERO_V8_DEFINITIONS. Do not guess pointer compression / sandbox ABI flags.")
  endif()
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/generated/core_prelude.h"
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/embed.py"
      "${CMAKE_CURRENT_SOURCE_DIR}/runtime/core.js"
      "${CMAKE_CURRENT_BINARY_DIR}/generated/core_prelude.h"
    DEPENDS runtime/core.js tools/embed.py VERBATIM)
  add_executable(zero native/main.cc native/host.cc
    "${CMAKE_CURRENT_BINARY_DIR}/generated/core_prelude.h")
  zero_compiler_options(zero)
  if(ZERO_BUILD_GRAPHICS)
    if(NOT WIN32 AND NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
      message(FATAL_ERROR "The EGL backend supports Linux and Windows ANGLE only")
    endif()
    add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/generated/graphics_prelude.h"
      COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/embed.py"
        "${CMAKE_CURRENT_SOURCE_DIR}/runtime/graphics.js"
        "${CMAKE_CURRENT_BINARY_DIR}/generated/graphics_prelude.h" kGraphicsPrelude
      DEPENDS runtime/graphics.js tools/embed.py VERBATIM)
    target_sources(zero PRIVATE native/graphics.cc "${CMAKE_CURRENT_BINARY_DIR}/generated/graphics_prelude.h")
    target_compile_definitions(zero PRIVATE ZERO_ENABLE_GRAPHICS)
    if(WIN32)
      target_sources(zero PRIVATE native/win32_window.cc)
      target_link_libraries(zero PRIVATE user32)
    endif()
    if(WIN32 AND ZERO_ANGLE_DIR)
      foreach(dll libEGL.dll libGLESv2.dll)
        if(NOT EXISTS "${ZERO_ANGLE_DIR}/${dll}")
          message(FATAL_ERROR "Missing standalone ANGLE DLL: ${ZERO_ANGLE_DIR}/${dll}")
        endif()
        add_custom_command(TARGET zero POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
          "${ZERO_ANGLE_DIR}/${dll}" "$<TARGET_FILE_DIR:zero>/${dll}" VERBATIM)
        install(FILES "${ZERO_ANGLE_DIR}/${dll}" DESTINATION bin)
      endforeach()
      if(EXISTS "${ZERO_ANGLE_DIR}/d3dcompiler_47.dll")
        add_custom_command(TARGET zero POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
          "${ZERO_ANGLE_DIR}/d3dcompiler_47.dll" "$<TARGET_FILE_DIR:zero>/d3dcompiler_47.dll" VERBATIM)
        install(FILES "${ZERO_ANGLE_DIR}/d3dcompiler_47.dll" DESTINATION bin)
      endif()
    endif()
  endif()
  target_include_directories(zero PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
  target_include_directories(zero SYSTEM PRIVATE "${ZERO_V8_INCLUDE_DIR}")
  target_compile_definitions(zero PRIVATE ${ZERO_V8_DEFINITIONS})
  if(EXISTS "${ZERO_V8_INCLUDE_DIR}/v8-gn.h")
    target_compile_definitions(zero PRIVATE V8_GN_HEADER)
  endif()
  if(MSVC)
    target_compile_options(zero PRIVATE /GR- /bigobj)
    target_link_libraries(zero PRIVATE winmm dbghelp advapi32 ws2_32)
  elseif(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(zero PRIVATE -fno-rtti)
  endif()
  target_link_libraries(zero PRIVATE zero_scheduler zero_platform "${ZERO_V8_LIBRARY}"
    ${ZERO_V8_EXTRA_LIBRARIES} Threads::Threads ${CMAKE_DL_LIBS})
  if(UNIX)
    target_link_libraries(zero PRIVATE m)
  endif()
  if(ZERO_V8_ICU_DATA)
    add_custom_command(TARGET zero POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
      "${ZERO_V8_ICU_DATA}" "$<TARGET_FILE_DIR:zero>/icudtl.dat" VERBATIM)
    install(FILES "${ZERO_V8_ICU_DATA}" DESTINATION bin)
  endif()
  install(TARGETS zero RUNTIME DESTINATION bin)
endif()
